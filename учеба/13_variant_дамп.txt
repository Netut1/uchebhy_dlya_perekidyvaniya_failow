using System;

namespace PhoneStationModel
{
    /// <summary>
    /// Виртуальные часы симуляции. Идут быстрее реальных.
    /// TimeFactor() — множитель активности в зависимости от времени суток.
    /// </summary>
    public class SimulationClock
    {
        public TimeSpan SimulatedTime { get; private set; } = TimeSpan.FromHours(8);

        /// <summary>Сколько симулированных минут проходит за 1 реальную секунду.</summary>
        public double MinutesPerSecond { get; set; } = 6.0;

        public int Hour => SimulatedTime.Hours;
        public int Minute => SimulatedTime.Minutes;
        public string TimeString => $"{Hour:D2}:{Minute:D2}";
        public bool IsDay => Hour >= 6 && Hour < 20;

        public void Tick(int elapsedMs, double speedMultiplier = 1.0)
        {
            double simMin = elapsedMs / 1000.0 * MinutesPerSecond * speedMultiplier;
            SimulatedTime = SimulatedTime.Add(TimeSpan.FromMinutes(simMin));
            if (SimulatedTime.TotalHours >= 24)
                SimulatedTime -= TimeSpan.FromHours(24);
        }

        /// <summary>Множитель «жизненной активности» по времени суток.</summary>
        public double TimeFactor()
        {
            int h = Hour;
            if (h >= 9 && h <= 18) return 1.5;   // рабочий день
            if (h >= 19 && h <= 22) return 1.2;  // вечер
            if (h >= 7 && h <= 8) return 1.0;    // утро
            if (h >= 23 || h <= 5) return 0.3;   // ночь
            return 0.7;                          // переходные часы
        }
    }
}

using _13_variant.Phones;

namespace PhoneStationModel
{
    public class Call
    {
        public CallSession Session { get; }
        public CallConnection Connection { get; }
        public Handset Caller => Connection.Caller;
        public Handset Callee => Connection.Callee;
        public bool IsIntercity { get; set; }

        public Call(Handset caller, Handset callee)
        {
            Session = new CallSession();
            Connection = new CallConnection(caller, callee);
        }

        public void Begin()
        {
            Session.Start();
            Caller.RaiseOutgoingCall(Callee);
            Session.MarkRinging();
            Callee.RaiseIncomingCall(Caller);
        }

        public void Answer()
        {
            Connection.Establish();
            Session.MarkActive();
            Caller.SetState(HandsetState.Talking);
            Callee.SetState(HandsetState.Talking);
        }

        public void End()
        {
            Connection.Drop();
            Session.End();
            Caller.RaiseCallEnded(Callee);
            Callee.RaiseCallEnded(Caller);
            if (Caller.State != HandsetState.Off) Caller.SetState(HandsetState.Idle);
            if (Callee.State != HandsetState.Off) Callee.SetState(HandsetState.Idle);
        }
    }
}

namespace PhoneStationModel
{
    public enum SessionState { Created, Dialing, Ringing, Active, Ended, Failed }

    public class CallSession
    {
        public Guid Id { get; } = Guid.NewGuid();
        public SessionState State { get; private set; } = SessionState.Created;
        public DateTime StartedAt { get; private set; }
        public DateTime? EndedAt { get; private set; }
        public TimeSpan Duration => (EndedAt ?? DateTime.Now) - StartedAt;

        public event EventHandler<SessionState>? StateChanged;

        public void Start()
        {
            StartedAt = DateTime.Now;
            Transition(SessionState.Dialing);
        }

        public void MarkRinging() => Transition(SessionState.Ringing);
        public void MarkActive() => Transition(SessionState.Active);
        public void End()
        {
            EndedAt = DateTime.Now;
            Transition(SessionState.Ended);
        }

        private void Transition(SessionState next)
        {
            if (State == next) return;
            State = next;
            StateChanged?.Invoke(this, next);
        }
    }
}

using PhoneStationModel;

namespace _13_variant.Phones
{
    public enum HandsetState { Off, Idle, Dialing, Ringing, Talking }

    public class Handset
    {
        private static int _nextId = 1;

        public int Id { get; } = _nextId++;
        public string Number { get; }
        public HandsetState State { get; protected set; } = HandsetState.Off;
        public PointF Position { get; set; }
        public bool IsRegistered { get; internal set; }

        public event EventHandler<CallEventArgs>? IncomingCall;
        public event EventHandler<CallEventArgs>? OutgoingCall;
        public event EventHandler<CallEventArgs>? CallEnded;
        public event EventHandler<Handset>? StateChanged;

        public Handset(string number) => Number = number;

        public virtual void SetState(HandsetState newState)
        {
            if (State == newState) return;
            State = newState;
            StateChanged?.Invoke(this, this);
        }

        public virtual bool CanMakeCall() => IsRegistered && State == HandsetState.Idle;

        public virtual void RaiseIncomingCall(Handset from) => IncomingCall?.Invoke(this, new CallEventArgs(from, this, "Входящий вызов"));
        public virtual void RaiseOutgoingCall(Handset to) => OutgoingCall?.Invoke(this, new CallEventArgs(this, to, "Исходящий вызов"));
        public virtual void RaiseCallEnded(Handset partner) => CallEnded?.Invoke(this, new CallEventArgs(this, partner, "Вызов завершён"));

        public override string ToString() => $"{GetType().Name} {Number} [{State}]";
    }
}

using _13_variant.Phones;

namespace PhoneStationModel
{
    public enum HookState { OnHook, OffHook }

    public class WiredHandset : Handset, IStationaryDevice
    {
        public string Location { get; }
        public DateTime InstalledAt { get; protected set; } = DateTime.Now;
        public TimeSpan Lifetime { get; protected set; }

        public HookState Hook { get; private set; } = HookState.OnHook;
        public bool HasMechanicalBell { get; set; } = true;
        public virtual bool HasCallerId => false;

        public event EventHandler<Handset>? BellRang;
        public event EventHandler<HookState>? HookChanged;

        public WiredHandset(string number, string location, TimeSpan lifetime) : base(number)
        {
            Location = location;
            Lifetime = lifetime;
        }

        public void ResetLifetime(TimeSpan lifetime)
        {
            InstalledAt = DateTime.Now;
            Lifetime = lifetime;
        }

        public void LiftHook()
        {
            if (Hook == HookState.OffHook) return;
            Hook = HookState.OffHook;
            HookChanged?.Invoke(this, Hook);
        }

        public void PutHook()
        {
            if (Hook == HookState.OnHook) return;
            Hook = HookState.OnHook;
            HookChanged?.Invoke(this, Hook);
            if (State is HandsetState.Talking or HandsetState.Dialing) SetState(HandsetState.Idle);
        }

        public override bool CanMakeCall() => base.CanMakeCall() && Hook == HookState.OffHook;

        public override void RaiseIncomingCall(Handset from)
        {
            if (HasMechanicalBell) BellRang?.Invoke(this, this);
            base.RaiseIncomingCall(from);
        }

        public override string ToString() => $"{base.ToString()} | Wired {Location} | Hook:{Hook}";
    }
}

using _13_variant.Phones;
using _13_variant.Station;

namespace PhoneStationModel
{
    public class CallRouter
    {
        private readonly RegistrationService _registry;
        private readonly Dictionary<Handset, Call> _active = new();
        private readonly List<Call> _history = new();
        private readonly Dictionary<Call, string> _intercityLines = new();
        private IntercityTrunk? _intercity;

        public event EventHandler<Call>? CallStarted;
        public event EventHandler<Call>? CallCompleted;
        public event EventHandler<CallEventArgs>? CallFailed;

        public CallRouter(RegistrationService registry) => _registry = registry;

        public void AttachIntercity(IntercityTrunk trunk) => _intercity = trunk;

        public IReadOnlyList<Call> ActiveCalls => _active.Values.Distinct().ToList();

        public Call? FindCallFor(Handset h) => _active.TryGetValue(h, out var c) ? c : null;

        public Call? Route(Handset from, string targetNumber)
        {
            if (targetNumber == IntercityTrunk.TrunkNumber || targetNumber.StartsWith("8-"))
                return RouteIntercity(from, targetNumber);

            if (!_registry.IsRegistered(from) || !from.CanMakeCall())
            {
                Fail(from, null, "Вызывающий абонент недоступен или не готов.");
                return null;
            }

            var to = _registry.Find(targetNumber);
            if (to == null)
            {
                Fail(from, null, $"Абонент {targetNumber} не найден.");
                return null;
            }

            if (_active.ContainsKey(from) || _active.ContainsKey(to))
            {
                Fail(from, to, "Один из абонентов уже занят.");
                return null;
            }

            var call = new Call(from, to);
            _active[from] = call;
            _active[to] = call;
            _history.Add(call);

            from.SetState(HandsetState.Dialing);
            call.Begin();

            CallStarted?.Invoke(this, call);
            return call;
        }

        public void AnswerCall(Handset who)
        {
            if (_active.TryGetValue(who, out var call) && !call.Connection.IsEstablished)
                call.Answer();
        }

        public void EndCall(Handset who)
        {
            if (_active.TryGetValue(who, out var call)) AbortCall(call);
        }

        public bool AbortCall(Call call)
        {
            if (!_active.ContainsKey(call.Caller)) return false;

            _active.Remove(call.Caller);
            _active.Remove(call.Callee);

            if (_intercityLines.Remove(call, out var lineId)) _intercity?.ReleaseLine(lineId);

            call.End();
            CallCompleted?.Invoke(this, call);
            return true;
        }

        public bool AbortLast()
        {
            var call = _history.LastOrDefault(c => _active.ContainsKey(c.Caller));
            return call != null && AbortCall(call);
        }

        public int AbortAll()
        {
            var calls = ActiveCalls.ToList();
            foreach (var c in calls) AbortCall(c);
            return calls.Count;
        }

        private void Fail(Handset from, Handset? to, string msg) => CallFailed?.Invoke(this, new CallEventArgs(from, to, msg));

        private Call? RouteIntercity(Handset from, string targetExternal)
        {
            if (_intercity == null || !_registry.IsRegistered(from) || !from.CanMakeCall() || _active.ContainsKey(from))
            {
                Fail(from, null, "Ошибка инициализации межгородского соединения.");
                return null;
            }

            var lineId = _intercity.RequestLine(from, targetExternal);
            if (lineId == null) return null;

            var virtualCallee = new ExternalSubscriber(lineId, targetExternal);
            var call = new Call(from, virtualCallee) { IsIntercity = true };

            _active[from] = call;
            _active[virtualCallee] = call;
            _intercityLines[call] = lineId;

            from.SetState(HandsetState.Dialing);
            call.Begin();
            call.Answer();

            CallStarted?.Invoke(this, call);
            return call;
        }
    }
}

using PhoneStationModel;

namespace _13_variant.Station
{
    public class IntercityTrunk : Handset, IStationaryDevice
    {
        public const string TrunkNumber = "8";

        public string Location => "Intercity";
        public DateTime InstalledAt { get; } = DateTime.Now;
        public TimeSpan Lifetime { get; } = TimeSpan.FromDays(3650);
        public int Capacity { get; }

        private readonly List<string> _activeLines = new();

        public event EventHandler<string>? LineAllocated;
        public event EventHandler<string>? LineReleased;
        public event EventHandler<CallEventArgs>? IntercityRejected;

        public IntercityTrunk(int capacity = 3) : base(TrunkNumber)
        {
            Capacity = capacity;
            SetState(HandsetState.Idle);
        }

        public bool HasFreeLine => _activeLines.Count < Capacity;
        public int ActiveLines => _activeLines.Count;

        public string? RequestLine(Handset from, string targetExternal)
        {
            if (!HasFreeLine)
            {
                IntercityRejected?.Invoke(this, new CallEventArgs(from, null, "Все каналы межгорода заняты."));
                return null;
            }

            var id = $"INT-{Guid.NewGuid().ToString("N").Substring(0, 6)}";
            _activeLines.Add(id);
            LineAllocated?.Invoke(this, $"{from.Number} → {targetExternal} [{id}]");
            return id;
        }

        public void ReleaseLine(string id)
        {
            if (_activeLines.Remove(id)) LineReleased?.Invoke(this, id);
        }
    }
}

using _13_variant.Phones;

namespace PhoneStationModel
{
    public class RegistrationService
    {
        private readonly Dictionary<string, Handset> _byNumber = new();
        private readonly List<Handset> _all = new();

        public event EventHandler<Handset>? Registered;
        public event EventHandler<Handset>? Unregistered;

        public IReadOnlyList<Handset> All => _all.AsReadOnly();

        public void Register(Handset h)
        {
            if (_byNumber.ContainsKey(h.Number)) throw new InvalidOperationException($"Номер {h.Number} занят.");
            _byNumber[h.Number] = h;
            _all.Add(h);
            h.IsRegistered = true;
            h.SetState(HandsetState.Idle);
            Registered?.Invoke(this, h);
        }

        public void Unregister(Handset h)
        {
            if (!_byNumber.Remove(h.Number)) return;
            _all.Remove(h);
            h.IsRegistered = false;
            h.SetState(HandsetState.Off);
            Unregistered?.Invoke(this, h);
        }

        public Handset? Find(string number) => _byNumber.TryGetValue(number, out var h) ? h : null;
        public bool IsRegistered(Handset h) => _byNumber.ContainsKey(h.Number);
    }
}

using _13_variant.Phones;

namespace PhoneStationModel
{
    public class EventOperator
    {
        private readonly ModelManager _model;
        private readonly TextBox _log;
        private readonly Action _invalidate;

        public EventOperator(ModelManager model, TextBox log, Action invalidate)
        {
            _model = model; _log = log; _invalidate = invalidate;
            Subscribe();
        }

        private void Subscribe()
        {
            _model.Log += (s, m) => Log(m);
            _model.Station.Registration.Registered += (s, h) => Log($"АТС: регистрация {h.Number}");
            _model.Station.Registration.Unregistered += (s, h) => Log($"АТС: снятие {h.Number}");
            _model.Station.Router.CallStarted += (s, c) => Log($"Маршрут: {c.Caller.Number} -> {c.Callee.Number}");
            _model.Station.Router.CallCompleted += (s, c) => Log($"Завершено: {c.Caller.Number} <-> {c.Callee.Number} ({c.Session.Duration.TotalSeconds:F1}с)");
            _model.Station.Router.CallFailed += (s, e) => Log($"Ошибка: {e.Message}");
            _model.HandsetAdded += (s, h) => SubscribeHandset(h);
            _model.HandsetRemoved += (s, h) => Log($"Удалён {h.Number}");
        }

        private void SubscribeHandset(Handset h)
        {
            h.StateChanged += (s, hs) => { Log($"{hs.Number}: -> {hs.State}"); _invalidate(); };
            h.IncomingCall += (s, e) => Log(h is WiredHandset { HasCallerId: false } ? $"{e.To?.Number}: входящий от Неизвестного" : $"{e.To?.Number}: входящий от {e.From.Number}");
            h.OutgoingCall += (s, e) => Log($"{e.From.Number}: исходящий к {e.To?.Number}");
            h.CallEnded += (s, e) => Log($"{e.From.Number}: завершение");

            if (h is WiredHandset wh) wh.BellRang += (s, self) => Log($"{self.Number}: звонок (механический)");
        }

        private void Log(string msg)
        {
            if (_log.InvokeRequired) { _log.Invoke(new Action(() => Log(msg))); return; }
            _log.AppendText($"{DateTime.Now:HH:mm:ss} {msg}{Environment.NewLine}");
            _invalidate();
        }
    }
}

            Size = new Size(1400, 800);

        }

            foreach (var w in _model.GetWires())

                using var pen = new Pen(w.Color, w.Thickness);
                g.DrawLine(pen, w.From, w.To);
            }

            var sp = _model.StationPosition;
            var triangle = new[]
            {
                new PointF(sp.X, sp.Y - 40),
                new PointF(sp.X - 55, sp.Y + 25),
                new PointF(sp.X + 55, sp.Y + 25)
            };
            g.FillPolygon(Brushes.DarkSlateGray, triangle);
            g.DrawPolygon(Pens.Black, triangle);

            var labelFont = new Font("Arial", 11, FontStyle.Bold);
            var labelSize = g.MeasureString("АТС", labelFont);
            g.DrawString("АТС", labelFont, Brushes.White,
                sp.X - labelSize.Width / 2f, sp.Y - 22);

                Color color;
                if (!h.IsRegistered) color = Color.Gray;

                    HandsetState.Dialing => Color.Orange,
                    HandsetState.Ringing => Color.Yellow,
                    HandsetState.Talking => Color.Red,

                using var brush = new SolidBrush(s.Color);
                g.FillEllipse(brush, pos.X - 6, pos.Y - 6, 12, 12);
            }
        }

using System;
using System.Collections.Generic;
using System.Drawing;
using System.Linq;
using _13_variant.Phones;

namespace PhoneStationModel
{
    public class ModelManager
    {
        private readonly Random _rand = new();
        private int _tickCounter;

        // ---------- СЕТКА ----------
        public int CellSize { get; set; } = 70;
        private PointF _gridOrigin = new(60, 60);
        private int _gridCols = 1, _gridRows = 1;
        private readonly HashSet<(int, int)> _occupiedCells = new();

        public BaseStation Station { get; } = new();
        public List<Handset> Handsets { get; } = new();
        public List<Signal> Signals { get; } = new();
        public PointF StationPosition { get; set; }
        public AtcHandset? AtcHandset { get; private set; }
        public IntercityTrunk? Intercity { get; private set; }
        public bool NeedsRepaint { get; private set; } = true;

        public event EventHandler<Handset>? HandsetAdded;
        public event EventHandler<Handset>? HandsetRemoved;
        public event EventHandler<string>? Log;

        public void MarkDirty() => NeedsRepaint = true;
        public void AfterRepaint() => NeedsRepaint = false;

        private TimeSpan RandomAtcLifetime() =>
            TimeSpan.FromSeconds(200 * (0.4 + _rand.NextDouble() * 0.2));

        private TimeSpan RandomLifetime() =>
            TimeSpan.FromSeconds(30 + _rand.Next(41));

        // ---------- ИНИЦИАЛИЗАЦИЯ ----------

        public void InitAtcHandset()
        {
            AtcHandset = new AtcHandset("000", RandomAtcLifetime())
            {
                Position = new PointF(StationPosition.X, StationPosition.Y + 25)
            };
            Attach(AtcHandset);
        }

        public void InitIntercity(int capacity = 3)
        {
            Intercity = new IntercityTrunk(capacity)
            {
                Position = new PointF(StationPosition.X + 90, StationPosition.Y - 50)
            };
            Station.Register(Intercity);
            Station.Router.AttachIntercity(Intercity);

            Intercity.LineAllocated += (s, msg) => Log?.Invoke(this, $"МЕЖГОРОД: {msg}");
            Intercity.LineReleased += (s, id) => Log?.Invoke(this, $"МЕЖГОРОД: линия {id} освобождена");
            Intercity.IntercityRejected += (s, e) => Log?.Invoke(this, $"МЕЖГОРОД: {e.Message}");
        }

        // ---------- ЖИЗНЕННЫЙ ЦИКЛ ----------

        private void Attach(Handset h)
        {
            try
            {
                Station.Register(h);
                Handsets.Add(h);
                HandsetAdded?.Invoke(this, h);
                MarkDirty();
            }
            catch (Exception ex) { Log?.Invoke(this, $"Ошибка: {ex.Message}"); }
        }

        private void RemoveHandset(Handset h, string reason)
        {
            ReleaseCell(h.Position);

            var call = Station.Router.FindCallFor(h);
            if (call != null) Station.Router.AbortCall(call);

            Station.Unregister(h);
            Handsets.Remove(h);
            HandsetRemoved?.Invoke(this, h);
            Log?.Invoke(this, $"Телефон {h.Number} удалён ({reason}).");
            MarkDirty();
        }

        // ---------- ДОБАВЛЕНИЕ ----------

        public Handset AddStationary(RectangleF b)
            => SetupDevice(new StationaryPhone(
                $"9{_rand.Next(1000000, 9999999)}", "Panasonic", "Office", RandomLifetime()), b);

        public Handset AddSmartphone(RectangleF b)
            => SetupDevice(new Smartphone(
                $"9{_rand.Next(1000000, 9999999)}", "Pixel"), b);

        public Handset AddWiredHandset(RectangleF b)
            => SetupDevice(new WiredHandset(
                $"9{_rand.Next(1000000, 9999999)}", "Hall", RandomLifetime()), b);

        public Handset AddPhoneBooth(RectangleF b)

            var pb = new PhoneBooth(

            pb.CoinsDepleted += (s, booth) =>
                Log?.Invoke(this, $"Будка {booth.Number}: монеты кончились, отбой.");
            return SetupDevice(pb, b);
        }

        public Handset AddRandom(RectangleF b) => _rand.Next(4) switch
        {
            0 => AddStationary(b),
            1 => AddSmartphone(b),
            2 => AddWiredHandset(b),
            _ => AddPhoneBooth(b)
        };

        // ---------- СЕТКА ----------

        public void ConfigureGrid(RectangleF bounds)
        {
            _gridOrigin = new PointF(bounds.Left, bounds.Top);
            _gridCols = Math.Max(1, (int)(bounds.Width / CellSize));
            _gridRows = Math.Max(1, (int)(bounds.Height / CellSize));
        }

        public bool TryFindFreeCell(out PointF worldPos)
        {
            int total = _gridCols * _gridRows;
            int attempts = Math.Min(total * 4, 300);

            for (int i = 0; i < attempts; i++)
            {
                int cx = _rand.Next(_gridCols);
                int cy = _rand.Next(_gridRows);
                if (_occupiedCells.Add((cx, cy)))
                {
                    worldPos = CellToWorld(cx, cy);
                    return true;
                }
            }

            for (int cx = 0; cx < _gridCols; cx++)
                for (int cy = 0; cy < _gridRows; cy++)
                    if (_occupiedCells.Add((cx, cy)))
                    {
                        worldPos = CellToWorld(cx, cy);
                        return true;
                    }

            worldPos = default;
            return false;
        }

        private PointF CellToWorld(int cx, int cy) =>
            new(_gridOrigin.X + cx * CellSize + CellSize / 2f,
                _gridOrigin.Y + cy * CellSize + CellSize / 2f);

        public void ReleaseCell(PointF pos)
        {
            int cx = (int)((pos.X - _gridOrigin.X) / CellSize);
            int cy = (int)((pos.Y - _gridOrigin.Y) / CellSize);
            _occupiedCells.Remove((cx, cy));
        }

        private Handset SetupDevice(Handset h, RectangleF bounds)
        {
            ConfigureGrid(bounds);

            if (!TryFindFreeCell(out var pos))
            {
                // Крайний случай: сетка заполнена — ставим случайно
                pos = new PointF(
                    _rand.Next((int)bounds.Left, (int)bounds.Right),
                    _rand.Next((int)bounds.Top, (int)bounds.Bottom));
            }

            h.Position = pos;
            Attach(h);
            return h;
        }

        // ---------- ВЫЗОВЫ ----------

        public void MakeRandomCall()
        {
            var idle = Handsets
                .Where(h => h != AtcHandset && h.IsRegistered && h.State == HandsetState.Idle)
                .ToList();
            if (idle.Count < 2) return;

            var from = idle[_rand.Next(idle.Count)];
            var to = idle.Where(h => h != from).ElementAt(_rand.Next(idle.Count - 1));

            if (from is WiredHandset wh) wh.LiftHook();

            if (Station.Call(from, to.Number) != null)
            {
                Signals.Add(new Signal
                {
                    From = from.Position,
                    Via = StationPosition,
                    To = to.Position,
                    OnComplete = () => Station.Answer(to)
                });
                MarkDirty();
            }
        }

        public void MakeIntercityCall()
        {
            var idle = Handsets
                .Where(h => h != AtcHandset && h.IsRegistered && h.State == HandsetState.Idle)
                .ToList();
            if (idle.Count == 0 || Intercity == null) return;

            var from = idle[_rand.Next(idle.Count)];
            if (from is WiredHandset wh) wh.LiftHook();

            if (Station.Call(from, $"8-{_rand.Next(100, 999)}-{_rand.Next(1000, 9999)}") != null)
            {
                Signals.Add(new Signal
                {
                    From = from.Position,
                    Via = StationPosition,
                    To = new PointF(StationPosition.X, 20),
                    Color = Color.MediumBlue
                });
                MarkDirty();
            }
        }

        public void AbortLastCall() =>
            Log?.Invoke(this, Station.Router.AbortLast() ? "Прерван последний вызов." : "Нет активных вызовов.");

        public void AbortAllCalls() =>
            Log?.Invoke(this, Station.Router.AbortAll() > 0 ? "Все вызовы прерваны." : "Нет активных вызовов.");

        // ---------- ТИК ----------

        public void Tick()
        {
            _tickCounter++;

            if (AtcHandset != null)
                AtcHandset.Position = new PointF(StationPosition.X, StationPosition.Y + 25);
            if (Intercity != null)
                Intercity.Position = new PointF(StationPosition.X + 90, StationPosition.Y - 50);

            if (AtcHandset != null && AtcHandset.IsExpired())
            {
                Station.Router.AbortAll();
                AtcHandset.ResetLifetime(RandomAtcLifetime());
            }

            foreach (var h in Handsets.ToList())
            {
                if (h == AtcHandset) continue;

                if (h is IStationaryDevice sd && sd.IsExpired())
                {
                    RemoveHandset(h, "истёк срок службы");
                    continue;
                }

                if (h is PhoneBooth booth
                    && booth.State == HandsetState.Talking
                    && _tickCounter % 80 == 0)
                    booth.ConsumeCoins(1);

                if (h is Phone p && p is not IStationaryDevice)
                {
                    bool inCall = p.State is HandsetState.Talking
                                           or HandsetState.Dialing
                                           or HandsetState.Ringing;
                    int period = inCall ? 10 : 40;
                    if (p is Smartphone sm)
                        period = Math.Max(2, (int)(period / sm.DrainMultiplier));

                    if (_tickCounter % period == 0) p.ConsumeBattery(1);
                    if (p.Battery == 0) RemoveHandset(h, "разряжен");
                }
            }

            for (int i = Signals.Count - 1; i >= 0; i--)
            {
                Signals[i].Progress += 0.02f;
                if (Signals[i].Progress >= 1f)
                {
                    Signals[i].OnComplete?.Invoke();
                    Signals.RemoveAt(i);
                }
                MarkDirty();
            }
        }

        // ---------- ПРОВОДА ----------

        public IEnumerable<Wire> GetWires()
        {
            var result = new List<Wire>();
            var drawnDirect = new HashSet<(int, int)>();
            bool intercityDrawn = false;

            foreach (var h in Handsets)
            {
                if (h == AtcHandset || !h.IsRegistered) continue;
                var call = Station.Router.FindCallFor(h);

                if (call != null && call.IsIntercity)
                {
                    result.Add(new Wire
                    {
                        From = h.Position,
                        To = StationPosition,
                        Color = Color.MediumBlue,
                        Thickness = 2.4f
                    });
                    if (!intercityDrawn)
                    {
                        result.Add(new Wire
                        {
                            From = StationPosition,
                            To = new PointF(StationPosition.X, 20),
                            Color = Color.MediumBlue,
                            Thickness = 2.4f
                        });
                        intercityDrawn = true;
                    }
                }
                else if (call != null && call.Connection.IsEstablished)
                {
                    var partner = call.Caller == h ? call.Callee : call.Caller;
                    if (partner is ExternalSubscriber) continue;

                    var key = h.Id < partner.Id ? (h.Id, partner.Id) : (partner.Id, h.Id);
                    if (drawnDirect.Add(key))
                        result.Add(new Wire
                        {
                            From = h.Position,
                            To = partner.Position,
                            Color = Color.SeaGreen,
                            Thickness = 2.5f
                        });
                }
                else if (call != null)
                    result.Add(new Wire
                    {
                        From = h.Position,
                        To = StationPosition,
                        Color = Color.Orange,
                        Thickness = 2f
                    });
                else if (h is IStationaryDevice)
                    result.Add(new Wire
                    {
                        From = h.Position,
                        To = StationPosition,
                        Color = Color.DimGray,
                        Thickness = 1.4f
                    });
            }
            return result;
        }
    }

    public class Signal
    {
        public PointF From { get; set; }
        public PointF Via { get; set; }
        public PointF To { get; set; }
        public float Progress { get; set; }
        public Color Color { get; set; } = Color.Red;
        public Action? OnComplete { get; set; }

        public PointF CurrentPosition => Progress < 0.5f
            ? Lerp(From, Via, Progress * 2f)
            : Lerp(Via, To, (Progress - 0.5f) * 2f);

        private static PointF Lerp(PointF a, PointF b, float t)
            => new(a.X + (b.X - a.X) * t, a.Y + (b.Y - a.Y) * t);
    }

using System;
using System.Drawing;

namespace PhoneStationModel
{
    /// <summary>
    /// Автономная имитация. Все события привязаны к виртуальному времени:
    /// в рабочие часы звонков больше, ночью — почти нет.
    /// </summary>
    public class SimulationEngine
    {
        private readonly ModelManager _model;
        private readonly Random _rand = new();

        public SimulationClock Clock { get; } = new();

        public bool Enabled { get; set; } = true;
        public double SpeedMultiplier { get; set; } = 1.0;

        // Настройки: сколько событий происходит за 1 СИМУЛИРОВАННЫЙ ЧАС
        public double DevicesPerHour { get; set; } = 1.0;
        public double LocalCallsPerHour { get; set; } = 2.0;
        public double IntercityPerHour { get; set; } = 0.8;
        public double AbortPerHour { get; set; } = 1.0;
        public int MaxDevices { get; set; } = 14;

        private double _devAcc, _callAcc, _interAcc, _abortAcc;

        public int TotalDevicesAdded { get; private set; }
        public int TotalCallsInitiated { get; private set; }
        public int TotalIntercityCalls { get; private set; }
        public int TotalAborts { get; private set; }

        public event EventHandler<string>? SimulationLog;
        public event EventHandler? StatsChanged;

        public SimulationEngine(ModelManager model) => _model = model;

        public void Tick(int elapsedMs)
        {
            // Часы идут даже на паузе — чтобы оператор видел время
            Clock.Tick(elapsedMs, SpeedMultiplier);

            if (!Enabled) return;

            // Сколько симулированных часов прошло за этот тик
            double simHours = elapsedMs / 1000.0
                              * Clock.MinutesPerSecond
                              * SpeedMultiplier
                              / 60.0;

            double factor = Clock.TimeFactor();

            _devAcc += simHours * DevicesPerHour * factor;
            _callAcc += simHours * LocalCallsPerHour * factor;
            _interAcc += simHours * IntercityPerHour * factor;
            _abortAcc += simHours * AbortPerHour * factor;

            var bounds = new RectangleF(60, 60, 800, 500);

            if (_devAcc >= 1.0 && _model.Handsets.Count < MaxDevices)
            {
                _devAcc -= 1.0;
                var added = _model.AddRandom(bounds);
                TotalDevicesAdded++;
                SimulationLog?.Invoke(this, $"Авто-добавлено: {added}");
                StatsChanged?.Invoke(this, EventArgs.Empty);
            }

            if (_callAcc >= 1.0)
            {
                _callAcc -= 1.0;
                _model.MakeRandomCall();
                TotalCallsInitiated++;
                StatsChanged?.Invoke(this, EventArgs.Empty);
            }

            if (_interAcc >= 1.0
                && _model.Intercity != null
                && _model.Intercity.HasFreeLine)
            {
                _interAcc -= 1.0;
                _model.MakeIntercityCall();
                TotalIntercityCalls++;
                StatsChanged?.Invoke(this, EventArgs.Empty);
            }

            if (_abortAcc >= 1.0)
            {
                _abortAcc -= 1.0;
                if (_model.Station.Router.ActiveCalls.Count > 0)
                {
                    _model.AbortLastCall();
                    TotalAborts++;
                    StatsChanged?.Invoke(this, EventArgs.Empty);
                }
            }
        }

        public void ResetStats()
        {
            TotalDevicesAdded = 0;
            TotalCallsInitiated = 0;
            TotalIntercityCalls = 0;
            TotalAborts = 0;
            StatsChanged?.Invoke(this, EventArgs.Empty);
        }
    }
}

        /// Круглый значок времени суток в стиле Minecraft:
        /// небо, солнце/луна на полукруге, время под ним.
        /// </summary>
        private void DrawClock(Graphics g, PointF c, float r)
        {
            var clock = _engine.Clock;
            int h = clock.Hour;

            Color sky = h switch
            {
                >= 6 and < 9 => Color.FromArgb(255, 200, 150),  // рассвет
                >= 9 and < 18 => Color.FromArgb(135, 206, 250),  // день
                >= 18 and < 21 => Color.FromArgb(255, 140, 100),  // закат
                _ => Color.FromArgb(20, 20, 60)      // ночь
            };

            using (var skyBrush = new SolidBrush(sky))
                g.FillEllipse(skyBrush, c.X - r, c.Y - r, r * 2, r * 2);
            g.DrawEllipse(Pens.Black, c.X - r, c.Y - r, r * 2, r * 2);

            bool isSun = h >= 6 && h < 18;
            double t;
            if (isSun)
                t = (h + clock.Minute / 60.0 - 6) / 12.0;
            else
            {
                double hh = h < 6 ? h + 6 : h - 18;
                t = (hh + clock.Minute / 60.0) / 12.0;
            }

            double angle = t * Math.PI;
            float px = c.X - (float)Math.Cos(angle) * (r * 0.7f);
            float py = c.Y - (float)Math.Sin(angle) * (r * 0.7f);

            if (isSun)
            {
                g.FillEllipse(Brushes.Gold, px - 9, py - 9, 18, 18);
                g.DrawEllipse(Pens.Orange, px - 9, py - 9, 18, 18);
            }
            else
            {
                g.FillEllipse(Brushes.LightYellow, px - 8, py - 8, 16, 16);
                g.DrawEllipse(Pens.Gray, px - 8, py - 8, 16, 16);
            }

            string text = clock.TimeString;
            var sz = g.MeasureString(text, _clockFont);
            Color tc = clock.IsDay ? Color.Black : Color.White;
            using var tb = new SolidBrush(tc);
            g.DrawString(text, _clockFont, tb, c.X - sz.Width / 2, c.Y + r + 4);
        }
