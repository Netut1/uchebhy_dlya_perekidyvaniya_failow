using System;

namespace PhoneStationModel
{
    /// <summary>
    /// Виртуальные часы симуляции. Идут быстрее реальных.
    /// TimeFactor() — множитель активности в зависимости от времени суток.
    /// </summary>
    public class SimulationClock
    {
        public TimeSpan SimulatedTime { get; private set; } = TimeSpan.FromHours(8);

        /// <summary>Сколько симулированных минут проходит за 1 реальную секунду.</summary>
        public double MinutesPerSecond { get; set; } = 6.0;

        public int Hour => SimulatedTime.Hours;
        public int Minute => SimulatedTime.Minutes;
        public string TimeString => $"{Hour:D2}:{Minute:D2}";
        public bool IsDay => Hour >= 6 && Hour < 20;

        public void Tick(int elapsedMs, double speedMultiplier = 1.0)
        {
            double simMin = elapsedMs / 1000.0 * MinutesPerSecond * speedMultiplier;
            SimulatedTime = SimulatedTime.Add(TimeSpan.FromMinutes(simMin));
            if (SimulatedTime.TotalHours >= 24)
                SimulatedTime -= TimeSpan.FromHours(24);
        }

        /// <summary>Множитель «жизненной активности» по времени суток.</summary>
        public double TimeFactor()
        {
            int h = Hour;
            if (h >= 9 && h <= 18) return 1.5;   // рабочий день
            if (h >= 19 && h <= 22) return 1.2;  // вечер
            if (h >= 7 && h <= 8) return 1.0;    // утро
            if (h >= 23 || h <= 5) return 0.3;   // ночь
            return 0.7;                          // переходные часы
        }
    }
}

using _13_variant.Phones;

namespace PhoneStationModel
{
    public sealed class Call
    {
        public CallSession Session { get; } = new();
        public CallConnection Connection { get; }
        public Handset Caller => Connection.Caller;
        public Handset Callee => Connection.Callee;

        public Call(Handset caller, Handset callee) => Connection = new CallConnection(caller, callee);
        public void Begin()
        {
            Session.Start(); Caller.RaiseOutgoingCall(Callee); Session.MarkRinging();
            Callee.SetState(HandsetState.Ringing); Callee.RaiseIncomingCall(Caller);
        }
        public void Answer()
        {
            Connection.Establish(); Session.MarkActive();
            Caller.SetState(HandsetState.Talking); Callee.SetState(HandsetState.Talking);
        }
        public void End()
        {
            Connection.Drop(); Session.End(); Caller.RaiseCallEnded(Callee); Callee.RaiseCallEnded(Caller);
            if (Caller.State != HandsetState.Off) Caller.SetState(HandsetState.Idle);
            if (Callee.State != HandsetState.Off) Callee.SetState(HandsetState.Idle);
        }
    }
}

namespace PhoneStationModel
{
    public enum SessionState { Created, Dialing, Ringing, Active, Ended, Failed }

    public class CallSession
    {
        public Guid Id { get; } = Guid.NewGuid();
        public SessionState State { get; private set; } = SessionState.Created;
        public DateTime StartedAt { get; private set; }
        public DateTime? EndedAt { get; private set; }
        public TimeSpan Duration => (EndedAt ?? DateTime.Now) - StartedAt;

        public event EventHandler<SessionState>? StateChanged;

        public void Start()
        {
            StartedAt = DateTime.Now;
            Transition(SessionState.Dialing);
        }

        public void MarkRinging() => Transition(SessionState.Ringing);
        public void MarkActive() => Transition(SessionState.Active);
        public void End()
        {
            EndedAt = DateTime.Now;
            Transition(SessionState.Ended);
        }

        private void Transition(SessionState next)
        {
            if (State == next) return;
            State = next;
            StateChanged?.Invoke(this, next);
        }
    }
}

namespace _13_variant.Phones
{
    public enum HandsetState { Off, Idle, Dialing, Ringing, Talking }

    public class Handset
    {
        private static int _nextId = 1;

        public int Id { get; } = _nextId++;
        public string Number { get; }
        public HandsetState State { get; protected set; } = HandsetState.Off;
        public PointF Position { get; set; }
        public bool IsRegistered { get; internal set; }

        public event EventHandler<CallEventArgs>? IncomingCall;
        public event EventHandler<CallEventArgs>? OutgoingCall;
        public event EventHandler<CallEventArgs>? CallEnded;
        public event EventHandler<Handset>? StateChanged;

        public Handset(string number) => Number = number;

        public virtual void SetState(HandsetState newState)
        {
            if (State == newState) return;
            State = newState;
            StateChanged?.Invoke(this, this);
        }

        public virtual bool CanMakeCall() => IsRegistered && State == HandsetState.Idle;

        public virtual void RaiseIncomingCall(Handset from) => IncomingCall?.Invoke(this, new CallEventArgs(from, this, "Входящий вызов"));
        public virtual void RaiseOutgoingCall(Handset to) => OutgoingCall?.Invoke(this, new CallEventArgs(this, to, "Исходящий вызов"));
        public virtual void RaiseCallEnded(Handset partner) => CallEnded?.Invoke(this, new CallEventArgs(this, partner, "Вызов завершён"));

        public override string ToString() => $"{GetType().Name} {Number} [{State}]";
    }
}

        {
            Model = model;
            Battery = battery;

        {
            Battery = Math.Max(0, Battery - amount);
            if (Battery == 0) SetState(HandsetState.Off);
        }

        public override bool CanMakeCall() => base.CanMakeCall() && Battery > 0;
        public override string ToString() => $"{base.ToString()} | {Model} | {Battery}%";

namespace PhoneStationModel
{
    public enum HookState { OnHook, OffHook }

    /// <summary>Квадратный абонентский аппарат с трубкой и рычагом.</summary>
    public class WiredHandset : Handset
    {
        public string Location { get; }
        public HookState Hook { get; private set; } = HookState.OnHook;
        public bool HasMechanicalBell { get; set; } = true;
        public event EventHandler<Handset>? BellRang;
        public event EventHandler<HookState>? HookChanged;

        public WiredHandset(string number, string location) : base(number) => Location = location;
        public void LiftHook()
        {
            if (Hook == HookState.OffHook) return;
            Hook = HookState.OffHook; HookChanged?.Invoke(this, Hook);
        }
        public void PutHook()
        {
            if (Hook == HookState.OnHook) return;
            Hook = HookState.OnHook; HookChanged?.Invoke(this, Hook);
            if (State is HandsetState.Talking or HandsetState.Dialing or HandsetState.Ringing) SetState(HandsetState.Idle);
        }
        public override bool CanMakeCall() => base.CanMakeCall() && Hook == HookState.OffHook;
        public override void RaiseIncomingCall(Handset from)
        {
            if (HasMechanicalBell) BellRang?.Invoke(this, this);
            base.RaiseIncomingCall(from);
        }
    }
}

using _13_variant.Phones;

namespace PhoneStationModel
{
    /// <summary>Маршрутизатор обычных абонентских вызовов. Межгорода в этой модели нет.</summary>
    public sealed class CallRouter
    {
        private readonly RegistrationService _registry;
        private readonly Dictionary<Handset, Call> _active = new();
        private readonly List<Call> _history = new();
        public event EventHandler<Call>? CallStarted;
        public event EventHandler<Call>? CallCompleted;
        public event EventHandler<CallEventArgs>? CallFailed;

        public CallRouter(RegistrationService registry) => _registry = registry;
        public IReadOnlyList<Call> ActiveCalls => _active.Values.Distinct().ToList();
        public Call? FindCallFor(Handset h) => _active.TryGetValue(h, out var call) ? call : null;

        public Call? Route(Handset from, string targetNumber)
        {
            if (!_registry.IsRegistered(from) || !from.CanMakeCall())
            {
                Fail(from, null, "Вызывающий абонент не зарегистрирован или не готов к звонку.");
                return null;
            }
            var to = _registry.Find(targetNumber);
            if (to == null) { Fail(from, null, $"Абонент {targetNumber} не найден."); return null; }
            if (ReferenceEquals(from, to)) { Fail(from, to, "Нельзя позвонить самому себе."); return null; }
            if (_active.ContainsKey(from) || _active.ContainsKey(to) || to.State != HandsetState.Idle)
            {
                Fail(from, to, "Один из абонентов уже занят.");
                return null;
            }

            var call = new Call(from, to);
            _active[from] = call; _active[to] = call; _history.Add(call);
            from.SetState(HandsetState.Dialing);
            call.Begin();
            CallStarted?.Invoke(this, call);
            return call;
        }

        public void AnswerCall(Handset who)
        {
            if (_active.TryGetValue(who, out var call) && !call.Connection.IsEstablished) call.Answer();
        }
        public void EndCall(Handset who) { if (_active.TryGetValue(who, out var call)) AbortCall(call); }

        public bool AbortCall(Call call)
        {
            if (!_active.ContainsKey(call.Caller)) return false;
            _active.Remove(call.Caller); _active.Remove(call.Callee);
            call.End(); CallCompleted?.Invoke(this, call); return true;
        }
        public bool AbortLast()
        {
            var call = _history.LastOrDefault(c => _active.ContainsKey(c.Caller));
            return call != null && AbortCall(call);
        }
        public int AbortAll()
        {
            var calls = ActiveCalls.ToList();
            foreach (var call in calls) AbortCall(call);
            return calls.Count;
        }
        private void Fail(Handset from, Handset? to, string message) => CallFailed?.Invoke(this, new CallEventArgs(from, to, message));
    }
}

using _13_variant.Phones;

namespace PhoneStationModel
{
    public class RegistrationService
    {
        private readonly Dictionary<string, Handset> _byNumber = new();
        private readonly List<Handset> _all = new();

        public event EventHandler<Handset>? Registered;
        public event EventHandler<Handset>? Unregistered;

        public IReadOnlyList<Handset> All => _all.AsReadOnly();

        public void Register(Handset h)
        {
            if (_byNumber.ContainsKey(h.Number)) throw new InvalidOperationException($"Номер {h.Number} занят.");
            _byNumber[h.Number] = h;
            _all.Add(h);
            h.IsRegistered = true;
            h.SetState(HandsetState.Idle);
            Registered?.Invoke(this, h);
        }

        public void Unregister(Handset h)
        {
            if (!_byNumber.Remove(h.Number)) return;
            _all.Remove(h);
            h.IsRegistered = false;
            h.SetState(HandsetState.Off);
            Unregistered?.Invoke(this, h);
        }

        public Handset? Find(string number) => _byNumber.TryGetValue(number, out var h) ? h : null;
        public bool IsRegistered(Handset h) => _byNumber.ContainsKey(h.Number);
    }
}

        private void SetAtcGeometry(RectangleF bounds)
        {
            float xInset = bounds.Width * 0.24f;
            float yInset = bounds.Height * 0.24f;

            PointF[] positions =
            {
                new(bounds.Left + xInset, bounds.Top + yInset),
                new(bounds.Right - xInset, bounds.Top + yInset),
                new(bounds.Right - xInset, bounds.Bottom - yInset),
                new(bounds.Left + xInset, bounds.Bottom - yInset)

                nodes = Atcs.ToArray();

            float radius = Math.Max(100f, Math.Min(bounds.Width, bounds.Height) * 0.43f);
            for (int i = 0; i < 4; i++)
            {
                nodes[i].Position = positions[i];
                nodes[i].VisibilityRadius = radius;
            }

        private Handset AddDevice(bool isPhone, bool permanent)
        {
            var number = (++_nextNumber).ToString();
            Handset device;
            if (isPhone) device = new Phone(number, "Телефон");
            else

                device = wired;

            device.Position = p;
            try
            {

                Log?.Invoke(this, $"Не удалось зарегистрировать абонента: {ex.Message}");
                return device;

            _deviceInfo[device] = new DeviceInfo
            {
                Permanent = permanent,
                RemainingHours = permanent ? double.PositiveInfinity : 6 + _rand.NextDouble() * 18,
                Vx = (float)(_rand.NextDouble() * 4.0 - 2.0),
                Vy = (float)(_rand.NextDouble() * 4.0 - 2.0)

            Handsets.Add(device);
            HandsetAdded?.Invoke(this, device);

            MarkDirty();
            return device;
        }

        private PointF FindSpawnPosition()
        {
            for (int attempt = 0; attempt < 150; attempt++)
            {
                PointF p;
                if (Atcs.Count == 4)
                {
                    var atc = Atcs[_rand.Next(Atcs.Count)];
                    double angle = _rand.NextDouble() * Math.PI * 2;
                    double radius = _rand.NextDouble() * Math.Max(30, atc.VisibilityRadius - 35);
                    p = new PointF(atc.Position.X + (float)(Math.Cos(angle) * radius), atc.Position.Y + (float)(Math.Sin(angle) * radius));
                    p = new PointF(Math.Clamp(p.X, _bounds.Left + 20, _bounds.Right - 20), Math.Clamp(p.Y, _bounds.Top + 20, _bounds.Bottom - 20));
                }
                else
                {
                    p = new PointF((float)(_bounds.Left + 25 + _rand.NextDouble() * Math.Max(1, _bounds.Width - 50)),
                                   (float)(_bounds.Top + 25 + _rand.NextDouble() * Math.Max(1, _bounds.Height - 50)));
                }
                if (Handsets.All(h => Distance(h.Position, p) >= 38)) return p;
            }
            return new PointF(_bounds.Left + 30, _bounds.Top + 30);
        }

        public void UpdateBounds(RectangleF bounds)
        {
            _bounds = bounds;
            if (Atcs.Count == 4) SetAtcGeometry(bounds);
            MarkDirty();
        }

        public void Tick(double elapsedSeconds, double elapsedVirtualHours)
        {
            foreach (var h in Handsets.ToList())
            {
                if (!_deviceInfo.TryGetValue(h, out var info)) continue;
                if (!info.Permanent)
                {
                    info.RemainingHours -= elapsedVirtualHours;
                    if (info.RemainingHours <= 0)
                    {
                        RemoveDevice(h, "истёк срок жизни");
                        continue;
                    }
                }

                // Случайное блуждание: скорость слегка меняется на каждом тике.
                info.Vx = Math.Clamp(info.Vx + (float)(_rand.NextDouble() - 0.5) * 0.75f, -2.8f, 2.8f);
                info.Vy = Math.Clamp(info.Vy + (float)(_rand.NextDouble() - 0.5) * 0.75f, -2.8f, 2.8f);
                h.Position = new PointF(h.Position.X + info.Vx * (float)(elapsedSeconds * 28),
                                        h.Position.Y + info.Vy * (float)(elapsedSeconds * 28));
                const float pad = 20f;
                if (h.Position.X < _bounds.Left + pad || h.Position.X > _bounds.Right - pad)
                {
                    info.Vx = -info.Vx;
                    h.Position = new PointF(Math.Clamp(h.Position.X, _bounds.Left + pad, _bounds.Right - pad), h.Position.Y);
                }
                if (h.Position.Y < _bounds.Top + pad || h.Position.Y > _bounds.Bottom - pad)
                {
                    info.Vy = -info.Vy;
                    h.Position = new PointF(h.Position.X, Math.Clamp(h.Position.Y, _bounds.Top + pad, _bounds.Bottom - pad));
                }
            }

            UpdateActiveRoutes();

            for (int i = Signals.Count - 1; i >= 0; i--)
            {
                Signals[i].Progress += (float)(elapsedSeconds * 3.0);
                if (Signals[i].Progress >= 1f)
                {
                    var signal = Signals[i];
                    Signals.RemoveAt(i);
                    signal.OnComplete?.Invoke();
                }
            }
            MarkDirty();
        }

        private static Color GetCallColor(Call call)
        {
            return call.Caller is Phone && call.Callee is Phone ? Color.DodgerBlue : Color.IndianRed;
        }

        public bool MakeRandomCall()
        {
            var idle = Handsets.Where(h => h.IsRegistered && h.State == HandsetState.Idle && (h.CanMakeCall() || h is WiredHandset)).ToList();
            if (idle.Count < 2) return false;
            var from = idle[_rand.Next(idle.Count)];
            var candidates = idle.Where(h => h != from).ToList();
            // Сначала выбираем адресата, у которого есть маршрут через доступные АТС.
            foreach (var to in candidates.OrderBy(_ => _rand.Next()))
            {
                var route = FindRoute(from.Position, to.Position);
                if (route == null || !CanReserve(route)) continue;
                if (from is WiredHandset wh && wh.Hook != HookState.OffHook) wh.LiftHook();
                var call = Station.Call(from, to.Number);
                if (call == null) continue;
                Reserve(call, route);
                _callRoutes[call] = route.ToList();
                CreateCallSignals(call, route);
                Log?.Invoke(this, $"Вызов: {from.Number} → {to.Number}; маршрут АТС {string.Join("–", route.Select(i => i + 1))}.");
                MarkDirty();
                return true;
            }
            Log?.Invoke(this, "Вызов не выполнен: нет свободного маршрута или абоненты вне зоны видимости АТС.");
            return false;
        }

        private List<int>? FindRoute(PointF from, PointF to)
        {
            var starts = Atcs.Where(a => Distance(a.Position, from) <= a.VisibilityRadius).Select(a => a.Id).ToList();
            var ends = Atcs.Where(a => Distance(a.Position, to) <= a.VisibilityRadius).Select(a => a.Id).ToList();
            if (starts.Count == 0 || ends.Count == 0) return null;

            var options = new List<List<int>>();
            foreach (int a in starts)
            foreach (int b in ends)
            {
                if (a == b) options.Add(new List<int> { a });
                else
                {
                    var clockwise = new List<int> { a };
                    for (int n = a, guard = 0; n != b && guard < 4; guard++) { n = (n + 1) % 4; clockwise.Add(n); }
                    var counter = new List<int> { a };
                    for (int n = a, guard = 0; n != b && guard < 4; guard++) { n = (n + 3) % 4; counter.Add(n); }
                    options.Add(clockwise);
                    options.Add(counter);
                }
            }
            return options.DistinctBy(r => string.Join(",", r)).OrderBy(r => r.Count).FirstOrDefault(CanReserve);
        }

        private void UpdateActiveRoutes()
        {
            foreach (var call in Station.Router.ActiveCalls.ToList())
            {
                var oldRoute = _callRoutes.TryGetValue(call, out var old) ? old : null;
                if (oldRoute != null && IsRouteStillUsable(call, oldRoute)) continue;

                ReleaseReservationOnly(call);
                var newRoute = FindRoute(call.Caller.Position, call.Callee.Position);
                if (newRoute == null)
                {
                    Station.Router.AbortCall(call);
                    Log?.Invoke(this, $"Звонок {call.Caller.Number} → {call.Callee.Number} прерван: абонент вне покрытия или нет свободного маршрута.");
                    continue;
                }
                Reserve(call, newRoute);
                _callRoutes[call] = newRoute.ToList();
                if (!call.Connection.IsEstablished) CreateCallSignals(call, newRoute);
                Log?.Invoke(this, $"Переключение маршрута звонка {call.Caller.Number} → {call.Callee.Number}: АТС {string.Join("–", newRoute.Select(i => i + 1))}.");
            }
        }

        private bool IsRouteStillUsable(Call call, List<int> route)
        {
            return Distance(call.Caller.Position, Atcs[route[0]].Position) <= Atcs[route[0]].VisibilityRadius
                && Distance(call.Callee.Position, Atcs[route[^1]].Position) <= Atcs[route[^1]].VisibilityRadius;
        }

        private void CreateCallSignals(Call call, List<int> route)
        {
            Signals.RemoveAll(s => s.RelatedCall == call);
            var points = new List<PointF> { call.Caller.Position };
            points.AddRange(route.Select(id => Atcs[id].Position));
            points.Add(call.Callee.Position);
            Color endpointColor = GetCallColor(call);
            for (int segment = 0; segment < points.Count - 1; segment++)
            {
                PointF a = points[segment], b = points[segment + 1];
                bool isAtcLink = segment > 0 && segment < points.Count - 2;
                PointF mid = new((a.X + b.X) / 2f, (a.Y + b.Y) / 2f);
                Signals.Add(new Signal
                {
                    From = a, Via = mid, To = b,
                    Color = isAtcLink ? Color.LimeGreen : endpointColor,
                    RelatedCall = call,
                    OnComplete = segment == points.Count - 2 ? () => Station.Answer(call.Callee) : null
                });
            }
        }

        private bool CanReserve(List<int> route)
        {
            if (route.Any(id => _stationLoads[id] >= StationCapacity)) return false;
            for (int i = 0; i < route.Count - 1; i++)
                if (_linkLoads[NormalizeLink(route[i], route[i + 1])] >= LinkCapacity) return false;
            return true;
        }

        private void Reserve(Call call, List<int> route)
        {
            var reservation = new RouteReservation { Stations = route.Distinct().ToList() };
            for (int i = 0; i < route.Count - 1; i++) reservation.Links.Add(NormalizeLink(route[i], route[i + 1]));
            foreach (int id in reservation.Stations) _stationLoads[id]++;
            foreach (var link in reservation.Links) _linkLoads[link]++;
            _reservations[call] = reservation;
            _callRoutes[call] = route.ToList();
        }

        private void ReleaseReservationOnly(Call call)
        {
            if (!_reservations.Remove(call, out var r)) return;
            foreach (int id in r.Stations) _stationLoads[id] = Math.Max(0, _stationLoads[id] - 1);
            foreach (var link in r.Links) _linkLoads[link] = Math.Max(0, _linkLoads[link] - 1);
            MarkDirty();
        }

        private void ReleaseRoute(Call call)
        {
            ReleaseReservationOnly(call);
            _callRoutes.Remove(call);
            Signals.RemoveAll(s => s.RelatedCall == call);
            MarkDirty();
        }

        private static (int, int) NormalizeLink(int a, int b) => a < b ? (a, b) : (b, a);
        private static float Distance(PointF a, PointF b) { float x = a.X - b.X, y = a.Y - b.Y; return (float)Math.Sqrt(x * x + y * y); }

        private void RemoveDevice(Handset h, string reason)
        {
            var call = Station.Router.FindCallFor(h);
            if (call != null) Station.Router.AbortCall(call);
            Station.Unregister(h);
            Handsets.Remove(h); _deviceInfo.Remove(h);
            HandsetRemoved?.Invoke(this, h);
            Log?.Invoke(this, $"Абонент №{h.Number} удалён: {reason}.");
            MarkDirty();
        }

        public IEnumerable<Wire> GetWires()
        {
            // Физическая сеть — квадрат без диагоналей.
            for (int i = 0; i < 4; i++)
                yield return new Wire { From = Atcs[i].Position, To = Atcs[(i + 1) % 4].Position, Color = Color.SeaGreen, Thickness = 3f, Kind = WireKind.StationLink };

            foreach (var call in Station.Router.ActiveCalls)
            {
                if (!_callRoutes.TryGetValue(call, out var route) || route.Count == 0) continue;
                Color endpointColor = GetCallColor(call);
                PointF previous = call.Caller.Position;
                yield return new Wire { From = previous, To = Atcs[route[0]].Position, Color = endpointColor, Thickness = 2.5f, Kind = WireKind.CallRoute };
                for (int i = 0; i < route.Count - 1; i++)
                    yield return new Wire { From = Atcs[route[i]].Position, To = Atcs[route[i + 1]].Position, Color = Color.LimeGreen, Thickness = 3f, Kind = WireKind.CallRoute };
                yield return new Wire { From = Atcs[route[^1]].Position, To = call.Callee.Position, Color = endpointColor, Thickness = 2.5f, Kind = WireKind.CallRoute };
            }
        }

        public int GetStationLoad(int id) => _stationLoads.TryGetValue(id, out int n) ? n : 0;
        public int GetLinkLoad(int a, int b) => _linkLoads.TryGetValue(NormalizeLink(a, b), out int n) ? n : 0;
        public double? GetRemainingHours(Handset h) => _deviceInfo.TryGetValue(h, out var info) && !info.Permanent ? info.RemainingHours : null;
        public bool IsPermanent(Handset h) => _deviceInfo.TryGetValue(h, out var info) && info.Permanent;
        public void MarkDirty() => NeedsRepaint = true;
        public void AfterRepaint() => NeedsRepaint = false;
    }

    public class Signal
    {
        public PointF From { get; set; }
        public PointF Via { get; set; }
        public PointF To { get; set; }
        public float Progress { get; set; }
        public Color Color { get; set; } = Color.DodgerBlue;
        public Call? RelatedCall { get; set; }
        public Action? OnComplete { get; set; }
        public PointF CurrentPosition => Progress < 0.5f ? Lerp(From, Via, Progress * 2f) : Lerp(Via, To, (Progress - 0.5f) * 2f);
        private static PointF Lerp(PointF a, PointF b, float t) => new(a.X + (b.X - a.X) * t, a.Y + (b.Y - a.Y) * t);
    }

namespace PhoneStationModel
{
    /// <summary>Управляет виртуальными часами и случайными попытками вызовов. Новые устройства автоматически не создаются.</summary>
    public sealed class SimulationEngine
    {
        private readonly ModelManager _model;
        private readonly Random _random = new();
        private double _callAccumulator;
        public SimulationClock Clock { get; } = new();
        public bool Enabled { get; set; }
        public double SpeedMultiplier { get; set; } = 1.0;
        public double CallsPerVirtualHour { get; set; } = 5.5;
        public event EventHandler? StatsChanged;
        public int TotalCallsInitiated { get; private set; }
        public SimulationEngine(ModelManager model) => _model = model;

        public void Tick(int elapsedMs)
        {
            double virtualHours = elapsedMs / 1000.0 * Clock.MinutesPerSecond * SpeedMultiplier / 60.0;
            Clock.Tick(elapsedMs, SpeedMultiplier);
            _model.Tick(elapsedMs / 1000.0, virtualHours);
            if (Enabled)
            {
                _callAccumulator += virtualHours * CallsPerVirtualHour * Clock.TimeFactor();
                while (_callAccumulator >= 1.0)
                {
                    _callAccumulator -= 1.0;
                    if (_model.MakeRandomCall()) TotalCallsInitiated++;
                }
            }
            StatsChanged?.Invoke(this, EventArgs.Empty);
        }
        public void ResetStats() { TotalCallsInitiated = 0; StatsChanged?.Invoke(this, EventArgs.Empty); }
    }
}
